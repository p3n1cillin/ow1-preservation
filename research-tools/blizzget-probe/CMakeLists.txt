cmake_minimum_required(VERSION 3.21)
project(blizzget_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "BlizzGet uses Windows WinHTTP; this probe requires Windows.")
endif()
set(BLIZZGET_SOURCE_DIR "" CACHE PATH "External checkout of d07RiV/blizzget")
if(NOT EXISTS "${BLIZZGET_SOURCE_DIR}/src/ngdp.cpp")
  message(FATAL_ERROR "Set BLIZZGET_SOURCE_DIR to a BlizzGet source checkout.")
endif()
set(CMAKE_CXX_STANDARD 14)
set(external_sources ngdp.cpp base/common.cpp base/http.cpp base/file.cpp base/path.cpp
    base/checksum.cpp base/string.cpp base/utf8.cpp base/pool.cpp base/error.cpp base/json.cpp base/regexp.cpp)
list(TRANSFORM external_sources PREPEND "${BLIZZGET_SOURCE_DIR}/src/")
add_executable(blizzget-probe probe.cpp ${external_sources})
target_include_directories(blizzget-probe PRIVATE "${BLIZZGET_SOURCE_DIR}/src" "${BLIZZGET_SOURCE_DIR}/libs")
target_link_directories(blizzget-probe PRIVATE "${BLIZZGET_SOURCE_DIR}/libs")
target_link_libraries(blizzget-probe PRIVATE winhttp shlwapi)
