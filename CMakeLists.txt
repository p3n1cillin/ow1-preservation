cmake_minimum_required(VERSION 3.21)
project(ow1_preservation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ixwebsocket CONFIG REQUIRED)
add_library(ow1_legacy_relay server/relay.cpp)
target_link_libraries(ow1_legacy_relay PUBLIC nlohmann_json::nlohmann_json)
target_include_directories(ow1_legacy_relay PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ow1-relay server/main.cpp)
target_link_libraries(ow1-relay PRIVATE ow1_legacy_relay ixwebsocket::ixwebsocket)
if(WIN32)
  target_link_libraries(ow1-relay PRIVATE bcrypt)
endif()
include(CTest)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_subdirectory(tools/offline-launch)
endif()
if(BUILD_TESTING)

    add_test(NAME startup-dump-inspection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/startup_dump.py)
    add_test(NAME disc-package-inspection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/disc_packages.py)
    add_test(NAME disc-resource-audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/disc_resource.py)
    add_test(NAME retail-bundle-bounds COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/retail_storage.py)
    add_test(NAME retail-stu-bounds COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/retail_stu.py)
    add_test(NAME retail-world-profile COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/retail_world.py)
    add_test(NAME bnet-probe-framing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bnet_probe.py)
    add_test(NAME instance-observer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/instance_probe.py)

  add_executable(protocol-tests tests/protocol_tests.cpp)
  target_link_libraries(protocol-tests PRIVATE ow1_legacy_relay)
  add_test(NAME protocol-regression COMMAND protocol-tests)
  add_executable(relay-integration tests/relay_integration.cpp)
  target_link_libraries(relay-integration PRIVATE ow1_legacy_relay ixwebsocket::ixwebsocket)
  if(WIN32)
    target_link_libraries(relay-integration PRIVATE bcrypt)
  endif()
  add_test(NAME relay-integration COMMAND relay-integration $<TARGET_FILE:ow1-relay>)
  set_tests_properties(relay-integration PROPERTIES TIMEOUT 30)
  if(WIN32)
    find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
    add_test(NAME client-inspection COMMAND ${POWERSHELL_EXECUTABLE} -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/inspect_client.ps1)
    if(TARGET ow1-offline-launch)
      add_executable(datagram-bridge-tests tests/datagram_bridge.cpp)
